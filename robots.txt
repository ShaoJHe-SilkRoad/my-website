# Keep crawlers off the reader-state URL variants (?depth=lazy&mode=STANDARD etc.).
# Real users still get them; the canonical tag tells search engines which URL counts.
Disallow: /*?*

# Stylesheets and scripts carry ?v= cache tokens; crawlers need them to render pages.
# The longer Allow patterns take precedence over the Disallow above.
Allow: /*.css?
Allow: /*.js?
